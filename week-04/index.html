00 / Loops

Loops

Repeated statements

scores = [72, 85, 91]
print(scores[0])
print(scores[1])
print(scores[2])
# Three items -> three print statements.

The same operation in a loop

scores = [72, 85, 91]
for score in scores:
    print(score)
print("Done")

The loop body

for score in [72, 85]:
    print("Score:", score)  # Runs twice.
    print(score >= 60)      # Also runs twice.
print("Finished")          # Runs once.

Zero iterations

for score in []:
    print(score)  # Never runs.
print("Finished")  # Still runs.

Pseudocode

SET remaining TO 3
WHILE remaining > 0
    OUTPUT remaining
    SET remaining TO remaining - 1
END WHILE
OUTPUT "Go"

01 / for over collections

for over collections

List items

scores = [72, 85, 91]
for score in scores:
    print(score)
print("Done")

Characters

for character in "CODE":
    print(character)
# C, O, D, E on separate lines.

Tuple unpacking

points = [(2, 3), (4, 1)]
for x, y in points:
    print(x + y)  # 5, then 5

Dictionary keys and values

student = {"name": "Ada", "score": 90}
for key in student:
    print(key)  # name, then score
for value in student.values():
    print(value)  # Ada, then 90

Dictionary items

scores = {"Ada": 90, "Lin": 75}
for name, score in scores.items():
    print(name, score)
# Ada 90
# Lin 75

Set order

skills = {"Python", "C", "SQL"}
# A set has no guaranteed iteration order.
for skill in sorted(skills):
    print(skill)  # C, Python, SQL

Trace the collection

  2. for score in scores:
  3. print(score)

02 / range()

range()

Start at zero

for number in range(4):
    print(number)  # 0, 1, 2, 3
# 4 is excluded.

Choose the start

for number in range(2, 5):
    print(number)  # 2, 3, 4

Positive steps

for number in range(2, 10, 2):
    print(number)  # 2, 4, 6, 8

Count backward

for number in range(5, 0, -1):
    print(number)  # 5, 4, 3, 2, 1

Empty ranges

print(list(range(3, 3)))      # []
print(list(range(5, 0, 1)))   # []
print(list(range(0, 5, -1)))  # []
# range(0, 5, 0) raises ValueError.

Items with positions

names = ["Ada", "Lin", "Sam"]
for index in range(len(names)):
    print(index, names[index])
# 0 Ada / 1 Lin / 2 Sam

Trace range()

Python

Bounds: −12 to 12. Step: −6 to 6; zero raises ValueError.

  2. print(number)
  3. print("Done")

03 / Counters and totals

Counters and totals

Total

total = 0                 # Initialize once.
for price in [30, 10, 25]:
    total += price        # 30 -> 40 -> 65
print(total)              # 65

Count matching items

count = 0
for score in [45, 60, 85, 32]:
    if score >= 60:
        count += 1
print(count)  # 2

Average and empty input

scores = [72, 85, 91]
total = 0
count = 0
for score in scores:
    total += score
    count += 1
print(total, count)
if count > 0:
    print(f"Average: {total / count:.1f}")
else:
    print("No scores")

Product

product = 1  # Multiplication starts at 1.
for factor in [2, 3, 4]:
    product *= factor
print(product)  # 24

Largest value

numbers = [-8, -3, -12]
largest = numbers[0]  # This list is not empty.
for number in numbers:
    if number > largest:
        largest = number
print(largest)  # -3

Keep the running totals

total = 0
running = []
for value in [5, 3, 7]:
    total += value
    running.append(total)
print(running)  # [5, 8, 15]

Collect passing scores

Transform every item

squares = []

Split into two groups

even = []
odd = []
for number in [1, 2, 3, 4]:

    else:

found = False

count = 0

lengths = {}

05 / while loops

while loops

Countdown

remaining = 3
while remaining > 0:
    print(remaining)
    remaining -= 1
print("Go")

Condition is checked first

remaining = 0
while remaining > 0:
    print(remaining)  # Does not run.
    remaining -= 1
print("Go")

Repeat until a sentinel

# input() returns these strings in order: "4", "6", "-1"
total = 0
value = int(input("Number (-1 to stop): "))
while value != -1:
    total += value
    value = int(input("Number (-1 to stop): "))
print(total)

Reach a target

value = 1
while value < 20:
    value *= 2
    print(value)  # 2, 4, 8, 16, 32

for: process each item

scores = [72, 85, 91]
for score in scores:
    print(score)

while: repeat while a condition holds

scores = [72, 85, 91]
index = 0
while index < len(scores):
    print(scores[index])
    index += 1  # Required for progress.

06 / break and continue

break and continue

break: stop the loop

names = ["Ada", "Lin", "Sam"]
for name in names:
    if name == "Lin":
        print("Found")
        break  # Sam is never visited.
print("Done")

continue: skip the remaining body

numbers = [4, -2, 7, 0, 9]
for number in numbers:
    if number < 0:
        continue
    print(number)
print("Done")

Search with a saved result

match = None
for score in [45, 72, 91]:
    if score >= 60:
        match = score
        break
print(match)  # 72, not 91

An unsuccessful search

found = False
for name in ["Ada", "Lin"]:
    if name == "Sam":
        found = True
        break
if not found:
    print("Not found")

Update before continue

number = 0
while number < 5:
    number += 1
    if number == 3:
        continue
    print(number)  # 1, 2, 4, 5

Stop at a sentinel

for value in [4, 6, -1, 99]:
    if value == -1:
        break
    print(value)  # 4, 6

Compare break and continue

Python

Up to 8 comma-separated integers from −20 to 20. Empty input is allowed.

  1. numbers = [4, -2, 7, 0, 9]
  2. for number in numbers:
  3. if number < 0:
  4. continue
  5. print(number)
  6. print("Done")

08 / Nested loops

Nested loops

Row and column

for row in range(1, 4):
    for column in range(1, 5):
        print(row, column, row * column)
print("Done")

All pairs

for color in ["red", "blue"]:
    for size in ["S", "M"]:
        print(color, size)
# red S / red M / blue S / blue M

Nested lists

matrix = [[1, 2], [3, 4]]
total = 0
for row in matrix:
    for value in row:
        total += value
print(total)  # 10

Count the body executions

count = 0
for row in range(3):
    for column in range(4):
        count += 1
print(count)  # 3 * 4 = 12

break exits only the inner loop

for row in range(2):
    for column in range(3):
        if column == 1:
            break
        print(row, column)
# 0 0
# 1 0

Print a small pattern

for row in range(1, 4):
    line = ""
    for column in range(row):
        line += "*"
    print(line)
# * / ** / *** on separate lines.

Trace row and column

Python
  1. for row in range(1, 4):
  2. for column in range(1, 5):
  3. print(row, column, row * column)
  4. print("Done")

09 / Worked programs

Worked programs

Score report

scores = [45, 60, 85, 90]
total = 0
passed = 0
for score in scores:
    total += score
    if score >= 60:
        passed += 1
print("Passed:", passed)
if len(scores) > 0:
    print(f"Average: {total / len(scores):.1f}")
else:
    print("No scores")

Shopping basket

basket = [
    {"name": "Notebook", "price": 30, "quantity": 2},
    {"name": "Pen", "price": 10, "quantity": 3},
]
total = 0
for item in basket:
    subtotal = item["price"] * item["quantity"]
    total += subtotal
    print(item["name"], subtotal)
print("Total:", total)

Number guessing

# Enter guesses from 1 to 10. At most three attempts.
secret = 7
attempts = 0
found = False
while attempts < 3 and not found:
    guess = int(input("Guess: "))
    attempts += 1
    if guess == secret:
        found = True
    elif guess < secret:
        print("Higher")
    else:
        print("Lower")
print("Found" if found else "No attempts left")

Count repeated values

counts = {}
for word in ["red", "blue", "red"]:
    counts[word] = counts.get(word, 0) + 1
print(counts)  # {'red': 2, 'blue': 1}
